cmake_minimum_required(VERSION 3.20)
project(r219_wgl_probe LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows-only WGL fixture")
endif()
add_executable(r219_wgl_probe probe.cpp)
target_compile_features(r219_wgl_probe PRIVATE cxx_std_17)
target_link_libraries(r219_wgl_probe PRIVATE opengl32 gdi32 user32)
if(MSVC)
  target_compile_options(r219_wgl_probe PRIVATE /W4 /WX)
endif()
